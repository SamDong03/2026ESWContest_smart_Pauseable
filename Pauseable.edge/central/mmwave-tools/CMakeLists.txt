cmake_minimum_required(VERSION 3.14)
project(pauseable_mmwave_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PAUSEABLE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PAUSEABLE_CENTRAL "${PAUSEABLE_ROOT}/Pauseable.edge/central")
add_subdirectory("${PAUSEABLE_CENTRAL}/mmwave" "${CMAKE_CURRENT_BINARY_DIR}/mmwave-core")
add_executable(mmwave-receiver main.cpp)
target_link_libraries(mmwave-receiver PRIVATE pauseable_mmwave)
include(CTest)
if(BUILD_TESTING)
    add_executable(mmwave-contract-test "${PAUSEABLE_CENTRAL}/tests/mmwave_contract_test.cpp")
    target_link_libraries(mmwave-contract-test PRIVATE pauseable_mmwave)
    add_test(NAME mmwave-contract COMMAND mmwave-contract-test)
    add_executable(mmwave-tracker-test "${PAUSEABLE_CENTRAL}/tests/mmwave_tracker_test.cpp")
    target_link_libraries(mmwave-tracker-test PRIVATE pauseable_mmwave)
    add_test(NAME mmwave-tracker COMMAND mmwave-tracker-test)
    add_executable(mmwave-archive-test "${PAUSEABLE_CENTRAL}/tests/mmwave_archive_test.cpp")
    target_link_libraries(mmwave-archive-test PRIVATE pauseable_mmwave)
    add_test(NAME mmwave-archive COMMAND mmwave-archive-test)
endif()
